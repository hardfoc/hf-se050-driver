78[[nodiscard]] inline Error ParseCommand(const std::uint8_t* capdu, std::size_t len, CommandFields* f) noexcept {
79 if (capdu == nullptr || f == nullptr || len < 4U) {
80 return Error::InvalidArgument;
81 }
82 *f = CommandFields{};
83 if (len == 4U) {
84 return Error::Ok; /* case 1 */
85 }
86 if (len == 5U) {
87 f->has_le = true; /* case 2S */
88 return Error::Ok;
89 }
90 if (capdu[4] != 0x00U) {
91 const std::size_t lc = capdu[4];
92 if (len == 5U + lc || len == 6U + lc) { /* case 3S / 4S */
93 f->data = capdu + 5U;
94 f->data_len = lc;
95 f->has_le = (len == 6U + lc);
96 return Error::Ok;
97 }
98 return Error::Protocol;
99 }
100 f->extended = true;
101 if (len == 7U) {
102 f->has_le = true; /* case 2E: 00 Le_hi Le_lo */
103 return Error::Ok;
104 }
105 const std::size_t lc = (static_cast<std::size_t>(capdu[5]) << 8) | capdu[6];
106 if (lc == 0U) {
107 return Error::Protocol;
108 }
109 if (len == 7U + lc || len == 9U + lc) { /* case 3E / 4E */
110 f->data = capdu + 7U;
111 f->data_len = lc;
112 f->has_le = (len == 9U + lc);
113 return Error::Ok;
114 }
115 return Error::Protocol;
116}

132[[nodiscard]] inline Error WrapCommand(ChannelState& st, const std::uint8_t* capdu,
133 std::size_t capdu_len, std::uint8_t* out,
134 std::size_t out_cap, std::size_t* out_len) noexcept {
135 if (out_len == nullptr) {
136 return Error::InvalidArgument;
137 }
138 *out_len = 0;
139 if (!st.open) {
140 return Error::NotSupported;
141 }
142 if (capdu == nullptr || out == nullptr || capdu_len < 4U) {
143 return Error::InvalidArgument;
144 }
145 CommandFields f{};
146 const Error pe = ParseCommand(capdu, capdu_len, &f);
147 if (pe != Error::Ok) {
148 return pe;
149 }
150
151 /* SE050 erratum JOS.2 (errata sheet Rev. 1.0, 3.13): inside platform SCP03 the SE050 does not advance
152 * its encryption counter for a command without C-DATA, and the next command fails. Workaround 3.13.3:
153 * append command data. Every command is therefore padded and encrypted, an empty one as a single
154 * `80 00 .. 00` block, so a case 1 command goes out as case 3 and a case 2 as case 4; the card
155 * decrypts the block back to no data and both counters advance. */
156 const std::size_t enc_len = ((f.data_len / 16U) + 1U) * 16U;
157 const std::size_t lc = enc_len + kMacFieldLen;
158 const bool extended = f.extended || lc > 255U;
159 if (lc > 0xFFFFU) {
160 return Error::InvalidArgument;
161 }
162 const std::size_t hdr = extended ? 7U : 5U;
163 const std::size_t le_len = f.has_le ? (extended ? 2U : 1U) : 0U;
164 if ((hdr + lc + le_len) > out_cap) {
165 return Error::BufferTooSmall;
166 }
167
168 out[0] = static_cast<std::uint8_t>(capdu[0] | kClaSmBit);
169 out[1] = capdu[1];
170 out[2] = capdu[2];
171 out[3] = capdu[3];
172 if (extended) {
173 out[4] = 0x00U;
174 out[5] = static_cast<std::uint8_t>((lc >> 8) & 0xFFU);
175 out[6] = static_cast<std::uint8_t>(lc & 0xFFU);
176 } else {
177 out[4] = static_cast<std::uint8_t>(lc);
178 }
179
180 {
181 if (f.data_len > 0U) {
182 std::memmove(out + hdr, f.data, f.data_len);
183 }
184 std::size_t padded = f.data_len;
185 if (!crypto::IsoPad80(out + hdr, &padded, enc_len) || padded != enc_len) {
186 crypto::SecureZero(out, out_cap);
187 return Error::BufferTooSmall;
188 }
189 std::uint8_t icv[16]{};
190 crypto::ScopedWipe wipe_icv(icv, sizeof(icv));
191 crypto::CommandIcv(st.senc, st.counter, icv);
192 if (!crypto::Aes128CbcEncrypt(st.senc, icv, out + hdr, out + hdr, enc_len)) {
193 crypto::SecureZero(out, out_cap);
194 return Error::InvalidArgument;
195 }
196 }
197
198 crypto::Cmac cm{};
199 crypto::CmacStart(cm, st.smac);
200 crypto::CmacUpdate(cm, st.mcv, 16U);
201 crypto::CmacUpdate(cm, out, hdr + enc_len);
202 std::uint8_t mac_full[16]{};
203 crypto::ScopedWipe wipe_mac(mac_full, sizeof(mac_full));
204 crypto::CmacFinish(cm, mac_full);
205 std::memcpy(out + hdr + enc_len, mac_full, kMacFieldLen);
206 std::memcpy(st.mcv, mac_full, 16);
207 std::size_t n = hdr + lc;
208 for (std::size_t i = 0; i < le_len; ++i) {
209 out[n++] = 0x00U;
210 }
211 *out_len = n;
212 return Error::Ok;
213}

230[[nodiscard]] inline Error UnwrapResponse(ChannelState& st, const std::uint8_t* rapdu,
231 std::size_t rapdu_len, std::uint8_t* out,
232 std::size_t out_cap, std::size_t* out_len) noexcept {
233 if (out_len == nullptr) {
234 return Error::InvalidArgument;
235 }
236 *out_len = 0;
237 if (!st.open) {
238 return Error::NotSupported;
239 }
240 if (rapdu == nullptr || out == nullptr) {
241 return Error::InvalidArgument;
242 }
243 if (rapdu_len < (kMacFieldLen + 2U)) {
244 WipeChannel(st);
245 return Error::Authentication;
246 }
247
248 const std::uint8_t sw1 = rapdu[rapdu_len - 2U];
249 const std::uint8_t sw2 = rapdu[rapdu_len - 1U];
250 const std::size_t enc_len = rapdu_len - kMacFieldLen - 2U;
251
252 crypto::Cmac cm{};
253 crypto::CmacStart(cm, st.srmac);
254 crypto::CmacUpdate(cm, st.mcv, 16U);
255 crypto::CmacUpdate(cm, rapdu, enc_len);
256 crypto::CmacUpdate(cm, rapdu + rapdu_len - 2U, 2U);
257 std::uint8_t mac_full[16]{};
258 crypto::ScopedWipe wipe_mac(mac_full, sizeof(mac_full));
259 crypto::CmacFinish(cm, mac_full);
260 if (!crypto::ConstantTimeEqual(mac_full, rapdu + enc_len, kMacFieldLen)) {
261 WipeChannel(st);
262 return Error::Authentication;
263 }
264
265 if (enc_len == 0U) {
266 crypto::IncCommandCounter(st.counter);
267 if (out_cap < 2U) {
268 return Error::BufferTooSmall;
269 }
270 out[0] = sw1;
271 out[1] = sw2;
272 *out_len = 2U;
273 return Error::Ok;
274 }
275 if ((enc_len % 16U) != 0U) {
276 WipeChannel(st);
277 return Error::Authentication;
278 }
279 if (enc_len > out_cap) {
280 crypto::IncCommandCounter(st.counter); /* answer consumed: stay in step with the card */
281 return Error::BufferTooSmall;
282 }
283
284 std::uint8_t icv[16]{};
285 crypto::ScopedWipe wipe_icv(icv, sizeof(icv));
286 crypto::ResponseIcv(st.senc, st.counter, icv);
287 std::size_t plain_len = 0;
288 if (!crypto::Aes128CbcDecrypt(st.senc, icv, rapdu, out, enc_len) ||
289 !crypto::IsoUnpad80(out, enc_len, &plain_len)) {
290 crypto::SecureZero(out, enc_len);
291 WipeChannel(st);
292 return Error::Authentication;
293 }
294 crypto::IncCommandCounter(st.counter);
295 if ((plain_len + 2U) > out_cap) {
296 crypto::SecureZero(out, enc_len);
297 return Error::BufferTooSmall;
298 }
299 out[plain_len] = sw1;
300 out[plain_len + 1U] = sw2;
301 *out_len = plain_len + 2U;
302 return Error::Ok;
303}

321inline bool FillPutKeyBlock(std::uint8_t* block, const std::uint8_t current_dek[crypto::kAes128KeyLen],
322 const std::uint8_t plain[crypto::kAes128KeyLen],
323 std::uint8_t kcv3[kPutKeyKcvLen]) noexcept {
324 if (block == nullptr || current_dek == nullptr || plain == nullptr || kcv3 == nullptr) {
325 return false;
326 }
327 block[0] = kPutKeyTypeAes;
328 block[1] = static_cast<std::uint8_t>(crypto::kAes128KeyLen + 1U);
329 block[2] = static_cast<std::uint8_t>(crypto::kAes128KeyLen);
330 crypto::Aes128EncryptBlock(current_dek, plain, block + 3U);
331 std::uint8_t ones[crypto::kAes128KeyLen];
332 std::memset(ones, 0x01, sizeof(ones));
333 std::uint8_t kcv_full[crypto::kAes128KeyLen]{};
334 crypto::Aes128EncryptBlock(plain, ones, kcv_full);
335 block[3U + crypto::kAes128KeyLen] = kPutKeyKcvLen;
336 std::memcpy(block + 3U + crypto::kAes128KeyLen + 1U, kcv_full, kPutKeyKcvLen);
337 std::memcpy(kcv3, kcv_full, kPutKeyKcvLen);
338 return true;
339}

349[[nodiscard]] inline Error BuildPutKey(std::uint8_t key_version,
350 const std::uint8_t current_dek[crypto::kAes128KeyLen],
351 const std::uint8_t new_enc[crypto::kAes128KeyLen],
352 const std::uint8_t new_mac[crypto::kAes128KeyLen],

354 std::uint8_t* capdu, std::size_t capdu_cap, std::size_t* capdu_len,

358 return Error::InvalidArgument;

360 *capdu_len = 0;
361 *expect_len = 0;
362 if (current_dek == nullptr || new_enc == nullptr || new_mac == nullptr || new_dek == nullptr ||
363 capdu == nullptr || expect_rsp == nullptr) {
364 return Error::InvalidArgument;
365 }
366 if (expect_cap < kPutKeyExpectRspBytes) {
367 return Error::BufferTooSmall;
368 }
369 std::uint8_t data[kPutKeyDataBytes]{};
370 std::size_t off = 0;
371 data[off++] = key_version;
372 expect_rsp[0] = key_version;
373 std::size_t kcv_off = 1;
374 if (!FillPutKeyBlock(data + off, current_dek, new_enc, expect_rsp + kcv_off)) {
375 return Error::InvalidArgument;
376 }
377 off += kPutKeyBlockBytes;
378 kcv_off += kPutKeyKcvLen;
379 if (!FillPutKeyBlock(data + off, current_dek, new_mac, expect_rsp + kcv_off)) {
380 return Error::InvalidArgument;
381 }
382 off += kPutKeyBlockBytes;
383 kcv_off += kPutKeyKcvLen;
384 if (!FillPutKeyBlock(data + off, current_dek, new_dek, expect_rsp + kcv_off)) {
385 return Error::InvalidArgument;
386 }
387 *expect_len = kPutKeyExpectRspBytes;
388 return apdu::BuildCaseShort(kClaGp, kInsPutKey, key_version, kPutKeyP2Multiple, data,
389 static_cast<std::uint8_t>(kPutKeyDataBytes), false, 0x00U, capdu, capdu_cap,

412 [[nodiscard]] Error OpenSecureChannel(T1Session<TransportT>& t1, const StaticKeys& keys,
413 std::uint32_t timeout_ms) noexcept {
414 using crypto::kAes128KeyLen;
415 using crypto::kChallengeLen;
416 using crypto::kCryptogramLen;
417 WipeChannel(st_);
418 failed_ = false;
419 if (!crypto::StaticKeysUsable(keys.host_static_key_enc, keys.host_static_key_enc_len,
420 keys.host_static_key_cmac, keys.host_static_key_cmac_len) ||
421 keys.host_challenge == nullptr || keys.host_challenge_len != kChallengeLen) {
422 return Error::NotSupported;
423 }
424
425 std::uint8_t capdu[16]{};
426 std::size_t capdu_len = 0;
427 Error e = apdu::BuildCaseShort(kClaGp, kInsInitializeUpdate, keys.key_version, 0x00U,
428 keys.host_challenge, static_cast<std::uint8_t>(kChallengeLen),
429 false, 0x00U, capdu, sizeof(capdu), &capdu_len);
430 if (e != Error::Ok) {
431 return e;
432 }
433
434 std::uint8_t rapdu[48]{};
435 std::size_t rapdu_len = 0;
436 e = t1.ExchangeInformation(capdu, capdu_len, rapdu, sizeof(rapdu), &rapdu_len, timeout_ms);
437 if (e != Error::Ok) {
438 return e;
439 }
440 if (rapdu_len < kInitUpdateRspMin) {
441 return Error::Protocol;
442 }
443 const std::uint8_t sw1 = rapdu[rapdu_len - 2U];
444 const std::uint8_t sw2 = rapdu[rapdu_len - 1U];
445 if (sw1 != 0x90U || sw2 != 0x00U) {
446 return Error::Protocol;
447 }
448
449 const std::uint8_t* card_challenge = rapdu + 13U;
450 const std::uint8_t* card_cryptogram = rapdu + 21U;
451
452 std::uint8_t senc[kAes128KeyLen]{};
453 std::uint8_t smac[kAes128KeyLen]{};
454 std::uint8_t srmac[kAes128KeyLen]{};
455 crypto::ScopedWipe wipe_senc(senc, sizeof(senc));
456 crypto::ScopedWipe wipe_smac(smac, sizeof(smac));
457 crypto::ScopedWipe wipe_srmac(srmac, sizeof(srmac));
458 if (!crypto::DeriveSessionKey(keys.host_static_key_enc, crypto::kDeriveSenc,
459 keys.host_challenge, card_challenge, senc) ||
460 !crypto::DeriveSessionKey(keys.host_static_key_cmac, crypto::kDeriveSmac,
461 keys.host_challenge, card_challenge, smac) ||
462 !crypto::DeriveSessionKey(keys.host_static_key_cmac, crypto::kDeriveSrmac,
463 keys.host_challenge, card_challenge, srmac)) {
464 return Error::InvalidArgument;
465 }
466
467 std::uint8_t expect_card[kCryptogramLen]{};
468 if (!crypto::ComputeCryptogram(smac, crypto::kDeriveCardCryptogram, keys.host_challenge,
469 card_challenge, expect_card) ||
470 !crypto::ConstantTimeEqual(expect_card, card_cryptogram, kCryptogramLen)) {
471 return Error::Protocol;
472 }
473
474 std::uint8_t host_cryptogram[kCryptogramLen]{};
475 if (!crypto::ComputeCryptogram(smac, crypto::kDeriveHostCryptogram, keys.host_challenge,
476 card_challenge, host_cryptogram)) {
477 return Error::InvalidArgument;
478 }
479
480 std::uint8_t ext[32]{};
481 ext[0] = kClaGpSecure;
482 ext[1] = kInsExternalAuthenticate;
483 ext[2] = kSecLevelCmacCencRmacRenc;
484 ext[3] = 0x00U;
485 ext[4] = 0x10U;
486 std::memcpy(ext + 5, host_cryptogram, kCryptogramLen);
487
488 std::uint8_t mcv[16]{};
489 std::uint8_t mac_in[16 + 13]{};
490 std::uint8_t mac_full[16]{};
491 crypto::ScopedWipe wipe_mac(mac_full, sizeof(mac_full));
492 std::memcpy(mac_in, mcv, 16);
493 std::memcpy(mac_in + 16, ext, 13);
494 crypto::Aes128Cmac(smac, mac_in, 16U + 13U, mac_full);
495 std::memcpy(ext + 13, mac_full, kCryptogramLen);
496 const std::size_t ext_len = 5U + 2U * kCryptogramLen;
497
498 std::uint8_t ext_rapdu[16]{};
499 std::size_t ext_rapdu_len = 0;
500 e = t1.ExchangeInformation(ext, ext_len, ext_rapdu, sizeof(ext_rapdu), &ext_rapdu_len,
501 timeout_ms);
502 if (e != Error::Ok) {
503 return e;
504 }
505 if (ext_rapdu_len < 2U || ext_rapdu[ext_rapdu_len - 2U] != 0x90U ||
506 ext_rapdu[ext_rapdu_len - 1U] != 0x00U) {
507 return Error::Protocol;
508 }
509
510 std::memcpy(st_.senc, senc, kAes128KeyLen);
511 std::memcpy(st_.smac, smac, kAes128KeyLen);
512 std::memcpy(st_.srmac, srmac, kAes128KeyLen);
513 std::memcpy(st_.mcv, mac_full, 16);
514 std::memset(st_.counter, 0, sizeof(st_.counter));
515 st_.counter[15] = 0x01U;
516 st_.open = true;
517 return Error::Ok;
518 }
