70 [[nodiscard]] Error TransmitApdu(const std::uint8_t* capdu, std::size_t capdu_len, std::uint8_t* rapdu_buf,
71 std::size_t rapdu_cap, std::size_t* rapdu_len,
72 std::uint32_t timeout_ms) noexcept {
73 if (rapdu_len == nullptr) {
74 return Error::InvalidArgument;
75 }
76 *rapdu_len = 0;
77 if (!EnsureInitialized()) {
78 return Error::NotInitialized;
79 }
80 if (scp03_.Failed()) {
81 return Error::Authentication; /* never fall back to plaintext after a lost channel */
82 }
83 if (!scp03_.IsOpen()) {
84 return t1_.ExchangeInformation(capdu, capdu_len, rapdu_buf, rapdu_cap, rapdu_len, timeout_ms);
85 }
86 if (scp03_wrap_ == nullptr || scp03_wrap_cap_ < scp03::kWrapScratchBytes) {
87 return Error::BufferTooSmall;
88 }
89 std::size_t wrapped_len = 0;
90 Error e = scp03_.WrapCommand(capdu, capdu_len, scp03_wrap_, scp03_wrap_cap_, &wrapped_len);
91 if (e != Error::Ok) {
92 return e; /* nothing sent; the channel is unchanged */
93 }
94 e = t1_.ExchangeInformation(scp03_wrap_, wrapped_len, rapdu_buf, rapdu_cap, rapdu_len, timeout_ms);
95 if (e != Error::Ok) {
96 /* The MAC chaining value moved on but the answer is lost: host and card are out of step. */
97 scp03_.Abort();
98 *rapdu_len = 0;
99 return e;
100 }
101 return scp03_.UnwrapResponse(rapdu_buf, *rapdu_len, rapdu_buf, rapdu_cap, rapdu_len);
102 }

429 [[nodiscard]] Error UnwrapAes256Rfc3394(const cmd::ObjectId& kek_id, const std::uint8_t* wrapped,
430 std::size_t wrapped_len, std::uint8_t* out, std::size_t out_cap,
431 std::size_t* out_len, std::uint32_t timeout_ms) noexcept {
432 struct Ctx {
433 Device* self;
434 cmd::ObjectId id;
435 std::uint32_t timeout_ms;
436 };
437 Ctx ctx{this, kek_id, timeout_ms};
438 auto decrypt = [](const std::uint8_t in[16], std::uint8_t outb[16], void* v) -> Error {
439 auto* c = static_cast<Ctx*>(v);
440 std::size_t n = 0;
441 const Error e = c->self->CipherAesEcb(c->id, true, in, 16U, outb, 16U, &n, c->timeout_ms);
442 if (e != Error::Ok) {
443 return e;
444 }
445 return (n == 16U) ? Error::Ok : Error::Protocol;
446 };
447 return rfc3394::Unwrap(wrapped, wrapped_len, out, out_cap, out_len, decrypt, &ctx);
448 }
