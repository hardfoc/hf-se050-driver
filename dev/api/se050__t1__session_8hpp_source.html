169 std::uint32_t timeout_ms) noexcept {
170 if (len_out == nullptr) {
171 return Error::InvalidArgument;
172 }
173 *len_out = 0;
174 ResetSequenceState();

176 if (se != Error::Ok) {
177 return se;

181 const Error re = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);

189 return Error::Protocol;

193 return Error::Protocol;

196 return Error::BufferTooSmall;

199 std::memcpy(buf, raw + 3U, ln);

204 if (atr::Parse(raw + 3U, ln, &profile) == Error::Ok && profile.ifsc != 0U) {
205 SetIfsc(profile.ifsc); /* honour the SE's block size for every later command */

220 [[nodiscard]] Error RecoverAfterHostReset(std::uint32_t timeout_ms) noexcept {
221 if (!transport_.EnsureInitialized()) {
222 return Error::NotInitialized;
223 }
224 std::uint8_t raw[t1::kMaxRawBlockBytes]{};
225 std::size_t raw_len = 0;
226 /* Ok or Crc: a stale block was drained. Transport or Timeout: nothing was pending. */
227 (void)RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
228 ResetSequenceState();
229 const Error se = SendSBlock(t1::kPcbSResyncReq, timeout_ms);
230 if (se != Error::Ok) {
231 return se;
232 }
233 std::uint8_t wtx_count = 0;
234 for (int guard = 0; guard < 16; ++guard) {
235 const Error re = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
236 if (re != Error::Ok) {
237 return re;
238 }
239 if (raw[0] != t1::kNadSeToHost) {
240 return Error::Protocol;
241 }
242 if (raw[1] == t1::kPcbSResyncResp) {
243 ResetSequenceState();
244 return Error::Ok;
245 }
246 if (raw[1] != t1::kPcbSWtxReq) {
247 return Error::Protocol;
248 }
249 const Error w = HandleSBlock(raw, raw_len, timeout_ms, &wtx_count);
250 if (w != Error::Ok) {
251 return w;
252 }
253 }
254 return Error::Protocol;

272 [[nodiscard]] Error ExchangeInformation(const std::uint8_t* cmd, std::size_t cmd_len, std::uint8_t* rsp,
273 std::size_t rsp_cap, std::size_t* rsp_len,
274 std::uint32_t timeout_ms) noexcept {
275 if (rsp_len == nullptr) {
276 return Error::InvalidArgument;
277 }
278 *rsp_len = 0;
279 if (!transport_.EnsureInitialized()) {
280 return Error::NotInitialized;
281 }
282
283 std::uint8_t retransmissions = 0;
284 std::size_t off = 0;
285 while (off < cmd_len) {
286 const std::size_t chunk = std::min(cmd_len - off, ifsc_);
287 const bool more_chain = (off + chunk) < cmd_len;
288 const std::uint8_t pcb = t1::IBlockPcb(host_ns_, more_chain);
289 const Error w = SendRawBlock(pcb, cmd + off, static_cast<std::uint8_t>(chunk), timeout_ms);
290 if (w != Error::Ok) {
291 return w;
292 }
293 host_ns_ ^= 1U;
294 off += chunk;
295 if (more_chain) {
296 const Error rr = ExpectChainAck(timeout_ms, &retransmissions);
297 if (rr != Error::Ok) {
298 return rr;
299 }
300 }
301 }
302
303 std::size_t total_rx = 0;
304 std::uint8_t wtx_count = 0;
305 for (int guard = 0; guard < 64; ++guard) {
306 std::uint8_t raw[t1::kMaxRawBlockBytes]{};
307 std::size_t raw_len = 0;
308 const Error rb = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
309 if (rb == Error::Crc) {
310 /* ISO/IEC 7816-3 11.6.3: ask for the block again with R(N(R) = expected, EDC error). */
311 if (++retransmissions > max_retransmissions_) {
312 return Error::Crc;
313 }
314 const Error n = SendRawBlock(t1::RBlockNackPcb(card_ns_, t1::kRBlockErrEdc), nullptr, 0U,
315 timeout_ms);
316 if (n != Error::Ok) {
317 return n;
318 }
319 continue;
320 }
321 if (rb != Error::Ok) {
322 return rb;
323 }
324 if (raw_len < 5U) {
325 return Error::Protocol;
326 }
327 if (raw[0] != t1::kNadSeToHost) {
328 return Error::Protocol;
329 }
330 if (t1::IsSBlock(raw[1])) {
331 const Error sb = HandleSBlock(raw, raw_len, timeout_ms, &wtx_count);
332 if (sb != Error::Ok) {
333 return sb;
334 }
335 continue;
336 }
337 if (t1::IsRBlock(raw[1])) {
338 /* The SE rejected our last I-block (N(R) = its N(S)): send it again. */
339 if (t1::RBlockNrFromPcb(raw[1]) != static_cast<std::uint8_t>(host_ns_ ^ 1U) ||
340 last_iblock_len_ == 0U) {
341 return Error::Protocol;
342 }
343 if (++retransmissions > max_retransmissions_) {
344 return Error::Protocol;
345 }
346 const Error rt = RetransmitLastIBlock(timeout_ms);
347 if (rt != Error::Ok) {
348 return rt;
349 }
350 continue;
351 }
352 const std::uint8_t seq = t1::IBlockSeqFromPcb(raw[1]);
353 if (seq != card_ns_) {
354 return Error::Sequence;
355 }
356 const std::uint8_t ln = raw[2];
357 if (3U + static_cast<std::size_t>(ln) + 2U != raw_len) {
358 return Error::Protocol;
359 }
360 if (total_rx + static_cast<std::size_t>(ln) > rsp_cap) {
361 return Error::BufferTooSmall;
362 }
363 if (ln > 0U) {
364 std::memcpy(rsp + total_rx, raw + 3U, ln);
365 total_rx += ln;
366 }
367 card_ns_ ^= 1U;
368 if (!t1::IBlockMore(raw[1])) {
369 *rsp_len = total_rx;
370 return Error::Ok;
371 }
372 const Error ack = SendRBlockAck(t1::RBlockAckPcb(card_ns_), timeout_ms);
373 if (ack != Error::Ok) {
374 return ack;
375 }
376 }
377 return Error::Protocol;
378 }

394 [[nodiscard]] Error SendRawBlock(std::uint8_t pcb, const std::uint8_t* inf, std::uint8_t inf_len,
395 std::uint32_t timeout_ms) noexcept {
396 if (static_cast<std::size_t>(inf_len) > t1::kMaxInformationField) {
397 return Error::InvalidArgument;
398 }
399 std::uint8_t frame[t1::kMaxRawBlockBytes]{};
400 frame[0] = t1::kNadHostToSe;
401 frame[1] = pcb;
402 frame[2] = inf_len;
403 if (inf_len > 0 && inf != nullptr) {
404 std::memcpy(frame + 3U, inf, inf_len);
405 }
406 crc::AppendEdc(frame, 3U + inf_len);
407 if (t1::IsIBlock(pcb)) {
408 std::memcpy(last_iblock_, frame, 3U + inf_len + 2U);
409 last_iblock_len_ = 3U + inf_len + 2U;
410 }
411 const Error e = transport_.Write(frame, 3U + inf_len + 2U, timeout_ms);
412 if (e != Error::Ok) {
413 return e;
414 }
415 transport_.delay_ms(inter_frame_delay_ms_);
416 return Error::Ok;
417 }

423 [[nodiscard]] Error RecvT1Block(std::uint8_t* out, std::size_t cap, std::size_t* total_len,
424 std::uint32_t timeout_ms) noexcept {
425 if (cap < 5U) {
426 return Error::BufferTooSmall;
427 }
428 Error e = ReadWithRetry(out, 3U, timeout_ms);
429 if (e != Error::Ok) {
430 return e;
431 }
432 const std::uint8_t ln = out[2];
433 if (3U + static_cast<std::size_t>(ln) + 2U > cap) {
434 return Error::BufferTooSmall;
435 }
436 if (ln + 2U > 0U) {
437 e = ReadWithRetry(out + 3U, ln + 2U, timeout_ms);
438 if (e != Error::Ok) {
439 return e;
440 }
441 }
442 const std::size_t tot = 3U + ln + 2U;
443 if (!crc::VerifyFrameCrc(out, tot)) {
444 return Error::Crc;
445 }
446 *total_len = tot;
447 return Error::Ok;
448 }

464 [[nodiscard]] Error ExpectChainAck(std::uint32_t timeout_ms, std::uint8_t* retransmissions) noexcept {
465 std::uint8_t wtx_count = 0;
466 for (int guard = 0; guard < 16; ++guard) {
467 std::uint8_t raw[t1::kMaxRawBlockBytes]{};
468 std::size_t raw_len = 0;
469 const Error e = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
470 if (e == Error::Crc) {
471 if (++(*retransmissions) > max_retransmissions_) {
472 return Error::Crc;
473 }
474 const Error n = SendRawBlock(t1::RBlockNackPcb(card_ns_, t1::kRBlockErrEdc), nullptr, 0U,
475 timeout_ms);
476 if (n != Error::Ok) {
477 return n;
478 }
479 continue;
480 }
481 if (e != Error::Ok) {
482 return e;
483 }
484 if (raw_len < 5U || raw[0] != t1::kNadSeToHost) {
485 return Error::Protocol;
486 }
487 if (t1::IsSBlock(raw[1])) {
488 const Error sb = HandleSBlock(raw, raw_len, timeout_ms, &wtx_count);
489 if (sb != Error::Ok) {
490 return sb;
491 }
492 continue;
493 }
494 if (!t1::IsRBlock(raw[1])) {
495 return Error::Protocol;
496 }
497 if (t1::RBlockNrFromPcb(raw[1]) == host_ns_) {
498 return Error::Ok; /* acknowledged: the SE expects the next N(S) */
499 }
500 if (++(*retransmissions) > max_retransmissions_) {
501 return Error::Protocol;
502 }
503 const Error rt = RetransmitLastIBlock(timeout_ms);
504 if (rt != Error::Ok) {
505 return rt;
506 }
507 }
508 return Error::Protocol;

515 [[nodiscard]] Error ReadWithRetry(std::uint8_t* rx, std::size_t rx_len, std::uint32_t timeout_ms) noexcept {
516 for (std::uint8_t attempt = 0; attempt <= max_read_retries_; ++attempt) {
517 const Error e = transport_.Read(rx, rx_len, timeout_ms);
518 if (e == Error::Ok) {
519 return Error::Ok;
520 }
521 if (!IsRetryableReadError(e) || attempt == max_read_retries_) {
522 return e;
523 }
524 transport_.delay_ms(read_retry_delay_ms_);
525 }
526 return Error::Transport;
527 }

529 [[nodiscard]] Error HandleSBlock(const std::uint8_t* raw, std::size_t raw_len, std::uint32_t timeout_ms,
530 std::uint8_t* wtx_count) noexcept {
531 if (raw == nullptr || raw_len < 5U) {
532 return Error::Protocol;
533 }
534 const std::uint8_t ln = raw[2];
535 if (3U + static_cast<std::size_t>(ln) + 2U != raw_len) {
536 return Error::Protocol;
537 }
538 if (raw[1] == t1::kPcbSWtxReq) {
539 if (wtx_count == nullptr || *wtx_count >= max_wtx_requests_) {
540 return Error::Protocol;
541 }
542 ++(*wtx_count);
543 return SendRawBlock(t1::kPcbSWtxResp, (ln > 0U) ? (raw + 3U) : nullptr, ln, timeout_ms);
544 }
545 if (raw[1] == t1::kPcbSResyncReq) {
546 ResetSequenceState();
547 return SendRawBlock(t1::kPcbSResyncResp, nullptr, 0U, timeout_ms);
548 }
549 return Error::Protocol;
550 }
