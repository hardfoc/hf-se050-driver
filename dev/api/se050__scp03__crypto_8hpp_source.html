148inline std::uint8_t Gmul(std::uint8_t a, std::uint8_t b) {
149 std::uint8_t p = 0;
150 for (int i = 0; i < 8; ++i) {
151 if ((b & 1U) != 0U) {
152 p = static_cast<std::uint8_t>(p ^ a);
153 }
154 const std::uint8_t hi = static_cast<std::uint8_t>(a & 0x80U);
155 a = static_cast<std::uint8_t>(a << 1);
156 if (hi != 0U) {
157 a = static_cast<std::uint8_t>(a ^ 0x1bU);
158 }
159 b = static_cast<std::uint8_t>(b >> 1);
160 }
161 return p;
162}

164inline void MixColumns(std::uint8_t s[16]) {
165 for (int c = 0; c < 4; ++c) {
166 std::uint8_t* col = s + 4 * c;
167 const std::uint8_t a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];
168 const std::uint8_t t = static_cast<std::uint8_t>(a0 ^ a1 ^ a2 ^ a3);
169 col[0] = static_cast<std::uint8_t>(a0 ^ t ^ Xtime(static_cast<std::uint8_t>(a0 ^ a1)));
170 col[1] = static_cast<std::uint8_t>(a1 ^ t ^ Xtime(static_cast<std::uint8_t>(a1 ^ a2)));
171 col[2] = static_cast<std::uint8_t>(a2 ^ t ^ Xtime(static_cast<std::uint8_t>(a2 ^ a3)));
172 col[3] = static_cast<std::uint8_t>(a3 ^ t ^ Xtime(static_cast<std::uint8_t>(a3 ^ a0)));
173 }
174}

176inline void InvMixColumns(std::uint8_t s[16]) {
177 for (int c = 0; c < 4; ++c) {
178 std::uint8_t* col = s + 4 * c;
179 const std::uint8_t a0 = col[0], a1 = col[1], a2 = col[2], a3 = col[3];
180 col[0] = static_cast<std::uint8_t>(Gmul(a0, 0x0e) ^ Gmul(a1, 0x0b) ^
181 Gmul(a2, 0x0d) ^ Gmul(a3, 0x09));
182 col[1] = static_cast<std::uint8_t>(Gmul(a0, 0x09) ^ Gmul(a1, 0x0e) ^
183 Gmul(a2, 0x0b) ^ Gmul(a3, 0x0d));
184 col[2] = static_cast<std::uint8_t>(Gmul(a0, 0x0d) ^ Gmul(a1, 0x09) ^
185 Gmul(a2, 0x0e) ^ Gmul(a3, 0x0b));
186 col[3] = static_cast<std::uint8_t>(Gmul(a0, 0x0b) ^ Gmul(a1, 0x0d) ^
187 Gmul(a2, 0x09) ^ Gmul(a3, 0x0e));
188 }

191inline void ExpandKey(const std::uint8_t key[16], std::uint8_t rk[176]) {
192 std::memcpy(rk, key, 16);
193 for (int i = 4; i < 44; ++i) {
194 std::uint8_t tmp[4] = {rk[4 * (i - 1)], rk[4 * (i - 1) + 1],
195 rk[4 * (i - 1) + 2], rk[4 * (i - 1) + 3]};
196 if ((i % 4) == 0) {
197 const std::uint8_t t0 = tmp[0];
198 tmp[0] = static_cast<std::uint8_t>(kSbox[tmp[1]] ^ kRcon[i / 4 - 1]);
199 tmp[1] = kSbox[tmp[2]];
200 tmp[2] = kSbox[tmp[3]];
201 tmp[3] = kSbox[t0];
202 }
203 rk[4 * i] = static_cast<std::uint8_t>(rk[4 * (i - 4)] ^ tmp[0]);
204 rk[4 * i + 1] = static_cast<std::uint8_t>(rk[4 * (i - 4) + 1] ^ tmp[1]);
205 rk[4 * i + 2] = static_cast<std::uint8_t>(rk[4 * (i - 4) + 2] ^ tmp[2]);
206 rk[4 * i + 3] = static_cast<std::uint8_t>(rk[4 * (i - 4) + 3] ^ tmp[3]);
207 }
208}

210inline void Aes128EncryptBlock(const std::uint8_t key[kAes128KeyLen],
211 const std::uint8_t in[16],
212 std::uint8_t out[16]) {
213 std::uint8_t rk[176];
214 ExpandKey(key, rk);
215 std::uint8_t s[16];
216 std::memcpy(s, in, 16);
217 AddRoundKey(s, rk);
218 for (int round = 1; round < 10; ++round) {
219 SubBytes(s);
220 ShiftRows(s);
221 MixColumns(s);
222 AddRoundKey(s, rk + 16 * round);
223 }
224 SubBytes(s);
225 ShiftRows(s);
226 AddRoundKey(s, rk + 160);
227 std::memcpy(out, s, 16);
228}

230inline void Aes128DecryptBlock(const std::uint8_t key[kAes128KeyLen],
231 const std::uint8_t in[16],
232 std::uint8_t out[16]) {
233 std::uint8_t rk[176];
234 ExpandKey(key, rk);
235 std::uint8_t s[16];
236 std::memcpy(s, in, 16);
237 AddRoundKey(s, rk + 160);
238 for (int round = 9; round >= 1; --round) {
239 InvShiftRows(s);
240 InvSubBytes(s);
241 AddRoundKey(s, rk + 16 * round);
242 InvMixColumns(s);
243 }
244 InvShiftRows(s);
245 InvSubBytes(s);
246 AddRoundKey(s, rk);
247 std::memcpy(out, s, 16);
248}

256inline void LeftShift1(std::uint8_t in[16]) {
257 std::uint8_t overflow = 0;
258 for (int i = 15; i >= 0; --i) {
259 const std::uint8_t next = static_cast<std::uint8_t>(in[i] >> 7);
260 in[i] = static_cast<std::uint8_t>((in[i] << 1) | overflow);
261 overflow = next;
262 }

265inline void GenerateSubkeys(const std::uint8_t key[16], std::uint8_t k1[16],
266 std::uint8_t k2[16]) {

269 Aes128EncryptBlock(key, z, L);
270 const std::uint8_t msb = static_cast<std::uint8_t>(L[0] >> 7);

284inline void Aes128Cmac(const std::uint8_t key[kAes128KeyLen],
285 const std::uint8_t* msg, std::size_t msg_len,
286 std::uint8_t out[kCmacLen]) {
287 std::uint8_t k1[16], k2[16];
288 GenerateSubkeys(key, k1, k2);
289 const std::size_t n = (msg_len == 0) ? 1 : ((msg_len + 15) / 16);
290 const bool complete = (msg_len != 0) && ((msg_len % 16) == 0);
291 std::uint8_t x[16]{};
292 std::uint8_t block[16]{};
293 for (std::size_t i = 1; i < n; ++i) {
294 std::memcpy(block, msg + 16 * (i - 1), 16);
295 XorBlock(block, x);
296 Aes128EncryptBlock(key, block, x);
297 }
298 std::memset(block, 0, 16);
299 const std::size_t last_off = (n - 1) * 16;
300 const std::size_t last_len = (msg_len == 0) ? 0 : (msg_len - last_off);
301 if (complete) {
302 std::memcpy(block, msg + last_off, 16);
303 XorBlock(block, k1);
304 } else {
305 if (msg != nullptr && last_len != 0) {
306 std::memcpy(block, msg + last_off, last_len);
307 }
308 block[last_len] = 0x80;
309 XorBlock(block, k2);
310 }
311 XorBlock(block, x);
312 Aes128EncryptBlock(key, block, out);
313}

316inline bool FillDerivationData(std::uint8_t constant, std::uint16_t bit_len,
317 std::uint8_t counter, const std::uint8_t* context,
318 std::uint16_t context_len, std::uint8_t* out,
319 std::size_t out_cap, std::size_t* out_len) {
320 if (out == nullptr || out_len == nullptr) {
321 return false;
322 }
323 const std::size_t need = kLabelLen + 4U + context_len;
324 if (need > out_cap) {
325 return false;
326 }
327 std::memset(out, 0, kLabelLen - 1U);
328 out[kLabelLen - 1U] = constant;
329 out[kLabelLen] = 0x00U;
330 out[kLabelLen + 1U] = static_cast<std::uint8_t>((bit_len >> 8) & 0xFFU);
331 out[kLabelLen + 2U] = static_cast<std::uint8_t>(bit_len & 0xFFU);
332 out[kLabelLen + 3U] = counter;
333 if (context_len != 0U && context != nullptr) {
334 std::memcpy(out + kLabelLen + 4U, context, context_len);
335 }
336 *out_len = need;
337 return true;
338}

386inline bool Aes128CbcEncrypt(const std::uint8_t key[kAes128KeyLen],
387 const std::uint8_t iv[16], const std::uint8_t* src,
388 std::uint8_t* dst, std::size_t len) {
389 if (key == nullptr || iv == nullptr || src == nullptr || dst == nullptr ||
390 (len % 16U) != 0U) {
391 return false;
392 }
393 std::uint8_t chain[16];
394 std::memcpy(chain, iv, 16);
395 for (std::size_t off = 0; off < len; off += 16U) {
396 std::uint8_t block[16];
397 std::memcpy(block, src + off, 16);
398 XorBlock(block, chain);
399 Aes128EncryptBlock(key, block, dst + off);
400 std::memcpy(chain, dst + off, 16);
401 }
402 return true;

426inline bool IsoPad80(std::uint8_t* buf, std::size_t* len, std::size_t cap) {
427 if (buf == nullptr || len == nullptr) {
428 return false;
429 }
430 if (*len >= cap) {
431 return false;
432 }
433 buf[(*len)++] = 0x80U;
434 while ((*len % 16U) != 0U) {
435 if (*len >= cap) {
436 return false;
437 }
438 buf[(*len)++] = 0x00U;

458inline void CmacStart(Cmac& c, const std::uint8_t key[kAes128KeyLen]) {
459 std::memcpy(c.key, key, kAes128KeyLen);
460 std::memset(c.x, 0, 16);
461 std::memset(c.buf, 0, 16);
462 c.buf_len = 0;
463 c.total = 0;
464}

466inline void CmacUpdate(Cmac& c, const std::uint8_t* data, std::size_t len) {
467 if (data == nullptr && len != 0U) {
468 return;
469 }
470 while (len > 0U) {
471 if (c.buf_len < 16U) {
472 const std::size_t take = (16U - c.buf_len) < len ? (16U - c.buf_len) : len;
473 std::memcpy(c.buf + c.buf_len, data, take);
474 c.buf_len += take;
475 c.total += take;
476 data += take;
477 len -= take;
478 }
479 if (c.buf_len == 16U && len > 0U) {
480 XorBlock(c.buf, c.x);
481 Aes128EncryptBlock(c.key, c.buf, c.x);
482 c.buf_len = 0;
483 }
484 }
485}

487inline void CmacFinish(Cmac& c, std::uint8_t out[kCmacLen]) {
488 std::uint8_t k1[16], k2[16];
489 GenerateSubkeys(c.key, k1, k2);
490 std::uint8_t block[16]{};
491 if (c.total == 0U) {
492 block[0] = 0x80U;
493 XorBlock(block, k2);
494 } else if (c.buf_len == 16U) {
495 std::memcpy(block, c.buf, 16);
496 XorBlock(block, k1);
497 } else {
498 std::memcpy(block, c.buf, c.buf_len);
499 block[c.buf_len] = 0x80U;
500 XorBlock(block, k2);
501 }
502 XorBlock(block, c.x);
503 Aes128EncryptBlock(c.key, block, out);
504}

531inline bool IsoUnpad80(const std::uint8_t* padded, std::size_t padded_len,
532 std::size_t* plain_len) {
533 if (padded == nullptr || plain_len == nullptr || padded_len < 16U ||
534 (padded_len % 16U) != 0U) {
535 return false;
536 }
537 std::size_t i = padded_len;
538 while (i > (padded_len - 16U) && i > 0U) {
539 if (padded[i - 1U] == 0x00U) {
540 --i;
541 continue;
542 }
543 if (padded[i - 1U] == 0x80U) {
544 *plain_len = i - 1U;
545 return true;
546 }
547 return false;
548 }
549 return false;
550}
