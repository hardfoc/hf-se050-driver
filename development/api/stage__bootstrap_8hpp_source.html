74inline void OnEvent(void* /*arg*/, esp_event_base_t base, std::int32_t id, void* data)
75{
76 if (base == WIFI_EVENT && id == WIFI_EVENT_STA_START) {
77 ESP_LOGI(kTag, "WiFi driver started — kicking off association.");
78 (void)esp_wifi_connect();
79 } else if (base == WIFI_EVENT && id == WIFI_EVENT_STA_DISCONNECTED) {
80 ESP_LOGW(kTag, "WiFi disconnected — retrying…");
81 (void)esp_wifi_connect();
82 } else if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
83 const auto* evt = static_cast<const ip_event_got_ip_t*>(data);
84 ESP_LOGI(kTag, "Got IP: " IPSTR, IP2STR(&evt->ip_info.ip));
85 xEventGroupSetBits(WifiEvents(), kWifiConnectedBit);
86 }
87}

90inline bool InitCoreServices()
91{
92 // NVS is required by the WiFi driver (calibration, station save).
93 esp_err_t err = nvs_flash_init();
94 if (err == ESP_ERR_NVS_NO_FREE_PAGES || err == ESP_ERR_NVS_NEW_VERSION_FOUND) {
95 (void)nvs_flash_erase();
96 err = nvs_flash_init();
97 }
98 if (err != ESP_OK) {
99 ESP_LOGE(kTag, "nvs_flash_init failed: %d", static_cast<int>(err));
100 return false;
101 }

115inline bool ConnectWifi()
116{
117 WifiEvents() = xEventGroupCreate();
118 if (WifiEvents() == nullptr) {
119 ESP_LOGE(kTag, "Cannot allocate WiFi event group.");
120 return false;
121 }
122
123 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
124 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
125
126 ESP_ERROR_CHECK(esp_event_handler_instance_register(
127 WIFI_EVENT, ESP_EVENT_ANY_ID, &OnEvent, nullptr, nullptr));
128 ESP_ERROR_CHECK(esp_event_handler_instance_register(
129 IP_EVENT, IP_EVENT_STA_GOT_IP, &OnEvent, nullptr, nullptr));
130
131 wifi_config_t wc{};
132 std::strncpy(reinterpret_cast<char*>(wc.sta.ssid), wifi::kSsid, sizeof(wc.sta.ssid) - 1);
133 std::strncpy(reinterpret_cast<char*>(wc.sta.password), wifi::kPassword, sizeof(wc.sta.password) - 1);
134 wc.sta.threshold.authmode = WIFI_AUTH_WPA2_PSK;
135
136 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
137 ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wc));
138 ESP_ERROR_CHECK(esp_wifi_start());
139
140 ESP_LOGI(kTag, "Associating to SSID \"%s\"…", wifi::kSsid);
141 const EventBits_t bits = xEventGroupWaitBits(WifiEvents(),
142 kWifiConnectedBit | kWifiFailedBit,
143 pdFALSE, pdFALSE,
144 pdMS_TO_TICKS(wifi::kAssociateTimeoutMs));
145 if ((bits & kWifiConnectedBit) != 0) {
146 ESP_LOGI(kTag, "WiFi link UP.");
147 return true;
148 }
149 ESP_LOGW(kTag, "WiFi association timed out / failed.");
150 return false;
151}
