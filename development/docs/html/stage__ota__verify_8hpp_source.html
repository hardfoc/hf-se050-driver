60template <class DeviceT>
61inline bool VerifyManifest(DeviceT& chip,
62 const std::uint8_t* image_hash, std::size_t hash_len,
63 const std::uint8_t* signature, std::size_t sig_len)
64{
65 bool is_valid = false;
66 const se050::Error ve = chip.EcdsaVerify(slot::kOtaVendorPublicKey,
67 se050::cmd::EcdsaAlgo::Sha256,
68 image_hash, hash_len,
69 signature, sig_len,
70 &is_valid, 800U);
71 if (ve != se050::Error::Ok) {
72 ESP_LOGE(kTag, "EcdsaVerify error: %u", static_cast<unsigned>(ve));
73 return false;
74 }
75 ESP_LOGI(kTag, "Manifest signature verdict: %s", is_valid ? "VALID" : "REJECTED");
76 return is_valid;
77}

107 ok ? "VALID" : "REJECTED");
108 ESP_LOGI(kTag, "==================== STAGE 5 — COMPLETE ========================");
