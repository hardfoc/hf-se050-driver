106template <class DeviceT>
107inline bool IsProvisioned(DeviceT& chip)
108{
109 bool exists = false;
110 if (chip.CheckObjectExists(slot::kProvisionedFlag, &exists, 300U) != se050::Error::Ok) {
111 return false; // Treat "can't query" as "not provisioned" — fail safe.
112 }
113 if (!exists) {
114 return false;
115 }
116
117 std::uint8_t value = 0U;
118 std::size_t out_len = 0U;
119 const se050::Error re = chip.ReadObject(slot::kProvisionedFlag,
120 /*use_offset=*/true, /*offset=*/0U,
121 /*use_length=*/true, /*length=*/1U,
122 &value, sizeof(value), &out_len, 300U);
123 return (re == se050::Error::Ok) && (out_len == 1U) && (value == 0x01U);
124}

220inline bool InstallCertificate(DeviceT& chip,
221 const std::uint8_t* cert_der, std::size_t cert_len)
222{
223 // If a cert is already stored, delete before rewriting (certificate rotation).
224 bool present = false;
225 (void)chip.CheckObjectExists(slot::kDeviceCertificate, &present, 300U);
226 if (present) {
227 ESP_LOGI(kTag, "[1e] Deleting existing device cert before rotation.");
228 (void)chip.DeleteSecureObject(slot::kDeviceCertificate, 400U);
229 }
230
231 const se050::Error we = chip.WriteBinary(slot::kDeviceCertificate,
232 cert_der, cert_len,
233 /*update=*/false, /*offset=*/0U,
234 /*last_chunk=*/true,
235 /*total_size=*/static_cast<std::uint16_t>(cert_len),
236 /*timeout_ms=*/500U);
237 if (we != se050::Error::Ok) {
238 ESP_LOGE(kTag, "[1e] Install device cert failed: %u", static_cast<unsigned>(we));
239 return false;
240 }
241 ESP_LOGI(kTag, "[1e] Device certificate installed (%u bytes).",
242 static_cast<unsigned>(cert_len));
243 return true;

313template <class DeviceT>
314inline bool RunStage(DeviceT& chip)
315{
316 ESP_LOGI(kTag, "==================== STAGE 1 — PROVISIONING ====================");
317
318 if (IsProvisioned(chip)) {
319 ESP_LOGI(kTag, "Device is already provisioned. Skipping factory stage.");
320 return true;
321 }
322
323 // -- 1a: keygen inside SE050 ------------------------------------------------
324 if (!CreateIdentityKeyPair(chip)) {
325 return false;
326 }
327
328 // -- 1b: export public component ------------------------------------------
329 std::uint8_t pubkey[192]{};
330 std::size_t pub_len = 0U;
331 (void)ExportPublicKey(chip, pubkey, sizeof(pubkey), &pub_len);
332
333 // -- 1c/d: CSR signing (stubbed digest) ------------------------------------
334 std::uint8_t csr_sig[128]{};
335 std::size_t csr_sig_len = 0U;
336 (void)SignCsrDigest(chip, csr_sig, sizeof(csr_sig), &csr_sig_len);
337
338 // -- 1e: install device cert (stubbed bytes) -------------------------------
339 // TODO(factory): Replace with real bytes returned by your CA.
340 constexpr std::array<std::uint8_t, 4> placeholder_cert = {
341 'C', 'R', 'T', '0'
342 };
343 (void)InstallCertificate(chip, placeholder_cert.data(), placeholder_cert.size());
344
345 // -- 1f: install AWS root CA (stubbed) -------------------------------------
346 // TODO(factory): Replace with the real Amazon ATS root PEM->DER bytes.
347 constexpr std::array<std::uint8_t, 4> placeholder_ca = {'C', 'A', '0', '0'};
348 (void)InstallServerRootCa(chip, placeholder_ca.data(), placeholder_ca.size());
349
350 // -- 1g: install OTA vendor public key (stubbed) ---------------------------
351 // TODO(factory): Replace with the actual vendor ECDSA P-256 public key
352 // (65 bytes uncompressed point 0x04 || X || Y).
353 std::array<std::uint8_t, 65> placeholder_vendor{};
354 placeholder_vendor[0] = 0x04U;
355 (void)InstallOtaVendorKey(chip, placeholder_vendor.data(), placeholder_vendor.size());
356
357 // -- 1g2: install the re-provisioning authority public key -----------------
358 // TODO(factory): Replace with the actual re-provisioning authority
359 // ECDSA P-256 public key (65 B). This key gates future factory-return
360 // resets; keep its private half in an offline HSM.
361 std::array<std::uint8_t, 65> placeholder_reprov{};
362 placeholder_reprov[0] = 0x04U;
363 (void)chip.WriteBinary(slot::kReprovisionAuthorityKey,
364 placeholder_reprov.data(), placeholder_reprov.size(),
365 false, 0U, true,
366 static_cast<std::uint16_t>(placeholder_reprov.size()), 500U);
367
368 // -- 1g3: initialize the re-provision monotonic counter to 0 --------------
369 constexpr std::array<std::uint8_t, 4> zero_ctr = {0U, 0U, 0U, 0U};
370 (void)chip.WriteBinary(slot::kReprovisionCounter,
371 zero_ctr.data(), zero_ctr.size(),
372 false, 0U, true, 4U, 400U);
373
374 // -- 1h: mark device as provisioned ----------------------------------------
375 if (!MarkProvisioned(chip)) {
376 return false;
377 }
378
379 ESP_LOGI(kTag, "==================== STAGE 1 — COMPLETE ========================");
380 return true;

383// =============================================================================
384// RE-PROVISIONING PATH
385// =============================================================================
386// Once a device is provisioned we intentionally make factory-return HARD.
387// The only software-accessible way to re-run Stage 1 is to present a
388// **signed re-provisioning token** that:
389//
390// 1. Carries a monotonically increasing counter > the NVM-stored counter
391// (prevents replay of a captured token).
392// 2. Embeds this device's unique ID (prevents moving a token between
393// chips).
394// 3. Is signed with the private half of `slot::kReprovisionAuthorityKey`
395// (prevents forgery by anyone except the offline HSM holder).
396//
397// If any check fails the sentinel stays put and the attempt is logged to
398// the cloud audit channel (see `stage_control.hpp`).
399//
400// **Token layout** (48 bytes + 72-byte ECDSA DER sig = 120 bytes total):
401// offset size field
402// 0 4 magic "HFRV"
403// 4 4 counter (LE, must be > NVM counter)
404// 8 18 device UID (SE050 unique ID; from GetUniqueId when
405// that API lands — for now we match against
406// a factory-programmed blob)
407// 26 22 reserved (zeros)
408// 48 var ECDSA-P256 DER signature over bytes [0..47]
409// =============================================================================
410
411inline constexpr std::uint32_t kReprovisionMagic = 0x56524648U; // 'HFRV' LE
412
416template <class DeviceT>
417inline bool ReadReprovisionCounter(DeviceT& chip, std::uint32_t* out_ctr)
418{
419 std::uint8_t bytes[4]{};
420 std::size_t got = 0U;
421 const se050::Error re = chip.ReadObject(slot::kReprovisionCounter,
422 true, 0U, true, 4U,
423 bytes, sizeof(bytes), &got, 300U);
424 if (re != se050::Error::Ok || got != 4U) {
425 return false;
426 }
427 *out_ctr = static_cast<std::uint32_t>(bytes[0])
428 | (static_cast<std::uint32_t>(bytes[1]) << 8)
429 | (static_cast<std::uint32_t>(bytes[2]) << 16)
430 | (static_cast<std::uint32_t>(bytes[3]) << 24);
431 return true;
432}

437template <class DeviceT>
438inline bool WriteReprovisionCounter(DeviceT& chip, std::uint32_t ctr)
439{
440 const std::array<std::uint8_t, 4> bytes = {
441 static_cast<std::uint8_t>(ctr & 0xFFU),
442 static_cast<std::uint8_t>((ctr >> 8) & 0xFFU),
443 static_cast<std::uint8_t>((ctr >> 16) & 0xFFU),
444 static_cast<std::uint8_t>((ctr >> 24) & 0xFFU),
445 };
446 const se050::Error we = chip.WriteBinary(slot::kReprovisionCounter,
447 bytes.data(), bytes.size(),
448 /*update=*/true, 0U, true, 4U, 400U);
449 return we == se050::Error::Ok;
450}

471template <class DeviceT>
472inline bool RequestReprovisioning(DeviceT& chip,
473 const std::uint8_t* token, std::size_t token_len)
474{
475 constexpr std::size_t kHeaderLen = 48U;
476 if (token == nullptr || token_len < kHeaderLen + 64U /*minimal sig*/) {
477 ESP_LOGW(kTag, "[reprov] Token too small (%u B).",
478 static_cast<unsigned>(token_len));
479 return false;
480 }
481
482 // -- 1) parse magic + counter ---------------------------------------------
483 const std::uint32_t magic = static_cast<std::uint32_t>(token[0])
484 | (static_cast<std::uint32_t>(token[1]) << 8)
485 | (static_cast<std::uint32_t>(token[2]) << 16)
486 | (static_cast<std::uint32_t>(token[3]) << 24);
487 if (magic != kReprovisionMagic) {
488 ESP_LOGW(kTag, "[reprov] Bad magic 0x%08X (want 0x%08X) — rejecting.",
489 static_cast<unsigned>(magic),
490 static_cast<unsigned>(kReprovisionMagic));
491 return false;
492 }
493 const std::uint32_t token_ctr = static_cast<std::uint32_t>(token[4])
494 | (static_cast<std::uint32_t>(token[5]) << 8)
495 | (static_cast<std::uint32_t>(token[6]) << 16)
496 | (static_cast<std::uint32_t>(token[7]) << 24);
497
498 // -- 2) replay protection --------------------------------------------------
499 std::uint32_t nvm_ctr = 0U;
500 if (!ReadReprovisionCounter(chip, &nvm_ctr)) {
501 ESP_LOGE(kTag, "[reprov] Cannot read NVM counter — abort.");
502 return false;
503 }
504 if (token_ctr <= nvm_ctr) {
505 ESP_LOGW(kTag, "[reprov] Stale token (tok=%u, nvm=%u) — replay attempt?",
506 static_cast<unsigned>(token_ctr), static_cast<unsigned>(nvm_ctr));
507 return false;
508 }
509
510 // -- 3) signature verification -------------------------------------------
511 std::array<std::uint8_t, 32> digest{};
512 (void)mbedtls_sha256(token, kHeaderLen, digest.data(), /*is224=*/0);
513
514 const std::uint8_t* sig_ptr = token + kHeaderLen;
515 const std::size_t sig_len = token_len - kHeaderLen;
516
517 bool sig_ok = false;
518 const se050::Error ve = chip.EcdsaVerify(slot::kReprovisionAuthorityKey,
519 se050::cmd::EcdsaAlgo::Sha256,
520 digest.data(), digest.size(),
521 sig_ptr, sig_len,
522 &sig_ok, 800U);
523 if (ve != se050::Error::Ok || !sig_ok) {
524 ESP_LOGE(kTag, "[reprov] Signature verify FAILED (err=%u, ok=%d).",
525 static_cast<unsigned>(ve), sig_ok ? 1 : 0);
526 return false;
527 }
528
529 // -- 4) commit: bump counter, clear sentinel ------------------------------
530 if (!WriteReprovisionCounter(chip, token_ctr)) {
531 ESP_LOGE(kTag, "[reprov] Failed to persist counter — refusing to clear.");
532 return false;
533 }
534 const se050::Error de = chip.DeleteSecureObject(slot::kProvisionedFlag, 500U);
535 if (de != se050::Error::Ok) {
536 ESP_LOGE(kTag, "[reprov] Could not delete sentinel: %u",
537 static_cast<unsigned>(de));
538 return false;
539 }
540
541 ESP_LOGW(kTag, "[reprov] Device RE-PROVISIONING authorized. Reboot to re-run Stage 1.");
542 return true;
543}
