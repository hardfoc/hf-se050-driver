357[[nodiscard]] inline Error ParseVersionInfo(const std::uint8_t* rapdu, std::size_t rapdu_len,
358 VersionInfo* out) noexcept {
359 if (out == nullptr) {
360 return Error::InvalidArgument;
361 }
362 const std::uint8_t* v = nullptr;
363 std::size_t v_len = 0;
364 apdu::StatusWords sw{};
365 const Error e = ParseResponseTag1(rapdu, rapdu_len, &v, &v_len, &sw);
366 if (e != Error::Ok) {
367 return e;
368 }
369 if (v_len < 7U) {
370 return Error::Protocol;
371 }
372 out->applet_major = v[0];
373 out->applet_minor = v[1];
374 out->applet_patch = v[2];
375 out->applet_config = static_cast<std::uint16_t>((static_cast<std::uint16_t>(v[3]) << 8U) | v[4]);
376 out->secure_box = static_cast<std::uint16_t>((static_cast<std::uint16_t>(v[5]) << 8U) | v[6]);
377 return Error::Ok;
378}
