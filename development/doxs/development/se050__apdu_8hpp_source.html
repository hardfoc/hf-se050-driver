44[[nodiscard]] inline Error BuildCase4Extended(std::uint8_t cla, std::uint8_t ins, std::uint8_t p1,
45 std::uint8_t p2, const std::uint8_t* data,
46 std::size_t data_len, std::uint8_t* out,
47 std::size_t out_cap, std::size_t* out_len) noexcept {
48 if (out_len == nullptr) {
49 return Error::InvalidArgument;
50 }
51 *out_len = 0;
52 /* SE050C2 on this desk answers short APDUs (SELECT 0x9000) and rejects
53 * Bertin-style extended Lc/Le for the same INS (GetVersion). Prefer
54 * short CASE 2/4 whenever the payload fits in one Lc byte. */
55 if (data_len <= 255U) {
56 if (data_len == 0U) {
57 if (out_cap < 5U) {
58 return Error::BufferTooSmall;
59 }
60 out[0] = cla;
61 out[1] = ins;
62 out[2] = p1;
63 out[3] = p2;
64 out[4] = 0x00U; /* CASE 2 Le */
65 *out_len = 5U;
66 return Error::Ok;
67 }
68 return BuildCaseShort(cla, ins, p1, p2, data, static_cast<std::uint8_t>(data_len), true, 0x00U,
69 out, out_cap, out_len);
70 }
71 constexpr std::size_t kHeader = 7U; /* CLA.. + 00 LcHi LcLo */
72 constexpr std::size_t kLeExt = 3U; /* 00 LeHi LeLo */
73 if (data_len > (kMaxApduCommandBytes - kHeader - kLeExt)) {
74 return Error::InvalidArgument;
75 }
76 const std::size_t need = kHeader + data_len + kLeExt;
77 if (need > out_cap) {
78 return Error::BufferTooSmall;
79 }
80 std::size_t i = 0;
81 out[i++] = cla;
82 out[i++] = ins;
83 out[i++] = p1;
84 out[i++] = p2;
85 out[i++] = 0x00;
86 out[i++] = static_cast<std::uint8_t>((data_len >> 8) & 0xFFU);
87 out[i++] = static_cast<std::uint8_t>(data_len & 0xFFU);
88 if (data_len > 0U && data != nullptr) {
89 std::memcpy(out + i, data, data_len);
90 i += data_len;
91 }
92 out[i++] = 0x00;
93 out[i++] = 0x00;
94 out[i++] = 0x00; /* Request up to 65536 bytes Ne (00 00 00) */
95 *out_len = i;
96 return Error::Ok;

105[[nodiscard]] inline Error BuildCaseShort(std::uint8_t cla, std::uint8_t ins, std::uint8_t p1,
106 std::uint8_t p2, const std::uint8_t* data,

143[[nodiscard]] inline Error ParseResponse(const std::uint8_t* rsp, std::size_t rsp_len,
144 const std::uint8_t** data_out, std::size_t* data_len_out,
145 StatusWords* sw_out) noexcept {
146 if (data_out == nullptr || data_len_out == nullptr || sw_out == nullptr) {
147 return Error::InvalidArgument;
148 }
149 if (rsp_len < 2U) {
150 return Error::Protocol;
151 }
152 *data_out = rsp;
153 *data_len_out = rsp_len - 2U;
154 sw_out->sw1 = rsp[rsp_len - 2U];
155 sw_out->sw2 = rsp[rsp_len - 1U];
156 return Error::Ok;
157}
