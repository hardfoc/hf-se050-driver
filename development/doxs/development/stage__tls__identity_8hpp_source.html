73template <class DeviceT>
74inline bool DemoSignHandshakeTranscript(DeviceT& chip)
75{
76 ESP_LOGI(kTag, "Simulating TLS CertificateVerify -> SE050 sign.");
77
78 // 1) Build an arbitrary 128-byte buffer that looks like a handshake
79 // transcript; hash it with SHA-256 to produce the 32-byte digest
80 // that a real TLS 1.2 stack would pass to `pk_sign()`.
81 std::array<std::uint8_t, 128> fake_transcript{};
82 for (std::size_t i = 0; i < fake_transcript.size(); ++i) {
83 fake_transcript[i] = static_cast<std::uint8_t>(i ^ 0x55U);
84 }
85 std::array<std::uint8_t, 32> hash{};
86 (void)mbedtls_sha256(fake_transcript.data(), fake_transcript.size(), hash.data(), /*is224=*/0);
87
88 // 2) Ask the SE050 to sign the digest.
89 std::uint8_t sig[128]{};
90 std::size_t sig_len = 0U;
91 const se050::Error sg = chip.EcdsaSign(slot::kDeviceIdentityKey,
92 se050::cmd::EcdsaAlgo::Sha256,
93 hash.data(), hash.size(),
94 sig, sizeof(sig), &sig_len, 600U);
95 if (sg != se050::Error::Ok) {
96 ESP_LOGE(kTag, "SE050 handshake-sign failed: %u", static_cast<unsigned>(sg));
97 return false;
98 }
99 ESP_LOGI(kTag, "Handshake signature produced (%u bytes DER).",
100 static_cast<unsigned>(sig_len));
101 return true;
102}

108inline void LogMbedTlsIntegrationRecipe()
109{
110 ESP_LOGI(kTag,
111 "RECIPE — wire SE050 into mbedTLS ECDSA sign:\n"
112 " 1) menuconfig: MBEDTLS_PK_ALT=y OR MBEDTLS_ECDSA_SIGN_ALT=y\n"
113 " 2) Provide mbedtls_pk_info_t::sign_func that calls\n"
114 " chip.EcdsaSign(slot::kDeviceIdentityKey, SHA256, hash, ..)\n"
115 " 3) mbedtls_pk_setup(&pk, &se050_pk_info);\n"
116 " mbedtls_ssl_conf_own_cert(&conf, &dev_cert, &pk);\n"
117 " 4) TLS 1.2/1.3 handshake now delegates to the SE050.\n"
118 " Full reference impl in docs/security_iot_ota_comms.md.");
