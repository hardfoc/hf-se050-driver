81[[nodiscard]] inline bool DecodeLength(const std::uint8_t* data, std::size_t data_len, std::size_t* off,
82 std::size_t* value_len) noexcept {
83 if (data == nullptr || off == nullptr || value_len == nullptr || *off >= data_len) {
84 return false;
85 }
86 const std::uint8_t l0 = data[(*off)++];
87 if ((l0 & 0x80U) == 0U) {
88 *value_len = l0;
89 return true;
90 }
91 const std::size_t num_bytes = static_cast<std::size_t>(l0 & 0x7FU);
92 if (num_bytes == 0U || num_bytes > 2U || *off + num_bytes > data_len) {
93 return false;
94 }
95 std::size_t len = 0;
96 for (std::size_t i = 0; i < num_bytes; ++i) {
97 len = (len << 8U) | data[(*off)++];
98 }
99 *value_len = len;
100 return true;
101}
