183 [[nodiscard]] Error ExchangeInformation(const std::uint8_t* cmd, std::size_t cmd_len, std::uint8_t* rsp,
184 std::size_t rsp_cap, std::size_t* rsp_len,
185 std::uint32_t timeout_ms) noexcept {
186 if (rsp_len == nullptr) {
187 return Error::InvalidArgument;
188 }
189 *rsp_len = 0;
190 if (!transport_.EnsureInitialized()) {
191 return Error::NotInitialized;
192 }
193
194 std::size_t off = 0;
195 std::uint8_t ns_host = host_ns_;
196 while (off < cmd_len) {
197 const std::size_t chunk =
198 std::min(cmd_len - off, static_cast<std::size_t>(t1::kMaxInformationField));
199 const bool more_chain = (off + chunk) < cmd_len;
200 const std::uint8_t pcb = t1::IBlockPcb(ns_host, more_chain);
201 const Error w = SendRawBlock(pcb, cmd + off, static_cast<std::uint8_t>(chunk), timeout_ms);
202 if (w != Error::Ok) {
203 return w;
204 }
205 ns_host ^= 1U;
206 off += chunk;
207 if (more_chain) {
208 const Error rr = ExpectRBlockFromCard(timeout_ms);
209 if (rr != Error::Ok) {
210 return rr;
211 }
212 }
213 }
214 host_ns_ = ns_host;
215
216 std::size_t total_rx = 0;
217 std::uint8_t wtx_count = 0;
218 for (int guard = 0; guard < 64; ++guard) {
219 std::uint8_t raw[t1::kMaxRawBlockBytes]{};
220 std::size_t raw_len = 0;
221 const Error rb = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
222 if (rb != Error::Ok) {
223 return rb;
224 }
225 if (raw_len < 5U) {
226 return Error::Protocol;
227 }
228 if (raw[0] != t1::kNadSeToHost) {
229 return Error::Protocol;
230 }
231 if (t1::IsSBlock(raw[1])) {
232 const Error sb = HandleSBlock(raw, raw_len, timeout_ms, &wtx_count);
233 if (sb != Error::Ok) {
234 return sb;
235 }
236 continue;
237 }
238 if (!t1::IsIBlock(raw[1])) {
239 return Error::Protocol;
240 }
241 const std::uint8_t seq = t1::IBlockSeqFromPcb(raw[1]);
242 if (seq != card_ns_) {
243 return Error::Sequence;
244 }
245 const std::uint8_t ln = raw[2];
246 if (3U + static_cast<std::size_t>(ln) + 2U != raw_len) {
247 return Error::Protocol;
248 }
249 if (total_rx + static_cast<std::size_t>(ln) > rsp_cap) {
250 return Error::BufferTooSmall;
251 }
252 if (ln > 0U) {
253 std::memcpy(rsp + total_rx, raw + 3U, ln);
254 total_rx += ln;
255 }
256 card_ns_ ^= 1U;
257 if (!t1::IBlockMore(raw[1])) {
258 *rsp_len = total_rx;
259 return Error::Ok;
260 }
261 const Error ack = SendRBlockAck(t1::RBlockAckPcb(card_ns_), timeout_ms);
262 if (ack != Error::Ok) {
263 return ack;
264 }
265 }
266 return Error::Protocol;
267 }

283 [[nodiscard]] Error SendRawBlock(std::uint8_t pcb, const std::uint8_t* inf, std::uint8_t inf_len,
284 std::uint32_t timeout_ms) noexcept {
285 if (static_cast<std::size_t>(inf_len) > t1::kMaxInformationField) {
286 return Error::InvalidArgument;
287 }
288 std::uint8_t frame[t1::kMaxRawBlockBytes]{};
289 frame[0] = t1::kNadHostToSe;
290 frame[1] = pcb;
291 frame[2] = inf_len;
292 if (inf_len > 0 && inf != nullptr) {
293 std::memcpy(frame + 3U, inf, inf_len);
294 }
295 crc::AppendEdc(frame, 3U + inf_len);
296 const Error e = transport_.Write(frame, 3U + inf_len + 2U, timeout_ms);
297 if (e != Error::Ok) {
298 return e;
299 }
300 transport_.delay_ms(inter_frame_delay_ms_);
301 return Error::Ok;
302 }

308 [[nodiscard]] Error RecvT1Block(std::uint8_t* out, std::size_t cap, std::size_t* total_len,
309 std::uint32_t timeout_ms) noexcept {
310 if (cap < 5U) {
311 return Error::BufferTooSmall;
312 }
313 Error e = ReadWithRetry(out, 3U, timeout_ms);
314 if (e != Error::Ok) {
315 return e;
316 }
317 const std::uint8_t ln = out[2];
318 if (3U + static_cast<std::size_t>(ln) + 2U > cap) {
319 return Error::BufferTooSmall;
320 }
321 if (ln + 2U > 0U) {
322 e = ReadWithRetry(out + 3U, ln + 2U, timeout_ms);
323 if (e != Error::Ok) {
324 return e;
325 }
326 }
327 const std::size_t tot = 3U + ln + 2U;
328 if (!crc::VerifyFrameCrc(out, tot)) {
329 return Error::Crc;
330 }
331 *total_len = tot;
332 return Error::Ok;
333 }

335 [[nodiscard]] Error ExpectRBlockFromCard(std::uint32_t timeout_ms) noexcept {
336 std::uint8_t wtx_count = 0;
337 for (int guard = 0; guard < 16; ++guard) {
338 std::uint8_t raw[t1::kMaxRawBlockBytes]{};
339 std::size_t raw_len = 0;
340 const Error e = RecvT1Block(raw, sizeof(raw), &raw_len, timeout_ms);
341 if (e != Error::Ok) {
342 return e;
343 }
344 if (raw_len < 5U || raw[0] != t1::kNadSeToHost) {
345 return Error::Protocol;
346 }
347 if (t1::IsSBlock(raw[1])) {
348 const Error sb = HandleSBlock(raw, raw_len, timeout_ms, &wtx_count);
349 if (sb != Error::Ok) {
350 return sb;
351 }
352 continue;
353 }
354 if (!t1::IsRBlock(raw[1])) {
355 return Error::Protocol;
356 }
357 return Error::Ok;
358 }
359 return Error::Protocol;
360 }

366 [[nodiscard]] Error ReadWithRetry(std::uint8_t* rx, std::size_t rx_len, std::uint32_t timeout_ms) noexcept {
367 for (std::uint8_t attempt = 0; attempt <= max_read_retries_; ++attempt) {
368 const Error e = transport_.Read(rx, rx_len, timeout_ms);
369 if (e == Error::Ok) {
370 return Error::Ok;
371 }
372 if (!IsRetryableReadError(e) || attempt == max_read_retries_) {
373 return e;
374 }
375 transport_.delay_ms(read_retry_delay_ms_);
376 }
377 return Error::Transport;
378 }

380 [[nodiscard]] Error HandleSBlock(const std::uint8_t* raw, std::size_t raw_len, std::uint32_t timeout_ms,
381 std::uint8_t* wtx_count) noexcept {
382 if (raw == nullptr || raw_len < 5U) {
383 return Error::Protocol;
384 }
385 const std::uint8_t ln = raw[2];
386 if (3U + static_cast<std::size_t>(ln) + 2U != raw_len) {
387 return Error::Protocol;
388 }
389 if (raw[1] == t1::kPcbSWtxReq) {
390 if (wtx_count == nullptr || *wtx_count >= max_wtx_requests_) {
391 return Error::Protocol;
392 }
393 ++(*wtx_count);
394 return SendRawBlock(t1::kPcbSWtxResp, (ln > 0U) ? (raw + 3U) : nullptr, ln, timeout_ms);
395 }
396 if (raw[1] == t1::kPcbSResyncReq) {
397 ResetSequenceState();
398 return SendRawBlock(t1::kPcbSResyncResp, nullptr, 0U, timeout_ms);
399 }
400 return Error::Protocol;
401 }
