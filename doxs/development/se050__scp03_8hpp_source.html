69[[nodiscard]] inline Error WrapCommand(ChannelState& st, const std::uint8_t* capdu,
70 std::size_t capdu_len, std::uint8_t* out,
71 std::size_t out_cap, std::size_t* out_len) noexcept {
72 if (out_len == nullptr) {
73 return Error::InvalidArgument;
74 }
75 *out_len = 0;
76 if (!st.open) {
77 return Error::NotSupported;
78 }
79 if (capdu == nullptr || out == nullptr || capdu_len < 4U) {
80 return Error::InvalidArgument;
81 }
82
83 const std::uint8_t cla = static_cast<std::uint8_t>(capdu[0] | kClaSmBit);
84 const std::uint8_t ins = capdu[1];
85 const std::uint8_t p1 = capdu[2];
86 const std::uint8_t p2 = capdu[3];
87 std::size_t data_len = 0;
88 const std::uint8_t* data = nullptr;
89 if (capdu_len > 4U) {
90 data_len = capdu[4];
91 if (data_len > 0U) {
92 if (capdu_len < (5U + data_len)) {
93 return Error::Protocol;
94 }
95 data = capdu + 5U;
96 }
97 }
98
99 std::size_t enc_len = 0;
100 if (data_len > 0U) {
101 if ((5U + data_len + 16U + kMacFieldLen) > out_cap) {
102 return Error::BufferTooSmall;
103 }
104 std::memcpy(out + 5U, data, data_len);
105 enc_len = data_len;
106 if (!crypto::IsoPad80(out + 5U, &enc_len, out_cap - 5U - kMacFieldLen)) {
107 return Error::BufferTooSmall;
108 }
109 std::uint8_t icv[16]{};
110 crypto::CommandIcv(st.senc, st.counter, icv);
111 if (!crypto::Aes128CbcEncrypt(st.senc, icv, out + 5U, out + 5U, enc_len)) {
112 return Error::InvalidArgument;
113 }
114 }
115
116 const std::size_t lc = enc_len + kMacFieldLen;
117 if (lc > 255U || (5U + lc) > out_cap) {
118 return Error::BufferTooSmall;
119 }
120 out[0] = cla;
121 out[1] = ins;
122 out[2] = p1;
123 out[3] = p2;
124 out[4] = static_cast<std::uint8_t>(lc);
125
126 crypto::Cmac cm{};
127 crypto::CmacStart(cm, st.smac);
128 crypto::CmacUpdate(cm, st.mcv, 16U);
129 crypto::CmacUpdate(cm, out, 5U + enc_len);
130 std::uint8_t mac_full[16]{};
131 crypto::CmacFinish(cm, mac_full);
132 std::memcpy(out + 5U + enc_len, mac_full, kMacFieldLen);
133 std::memcpy(st.mcv, mac_full, 16);
134 *out_len = 5U + lc;
135 return Error::Ok;
136}

143[[nodiscard]] inline Error UnwrapResponse(ChannelState& st, const std::uint8_t* rapdu,
144 std::size_t rapdu_len, std::uint8_t* out,
145 std::size_t out_cap, std::size_t* out_len) noexcept {
146 if (out_len == nullptr) {
147 return Error::InvalidArgument;
148 }
149 *out_len = 0;
150 if (!st.open) {
151 return Error::NotSupported;
152 }
153 if (rapdu == nullptr || out == nullptr || rapdu_len < 2U) {
154 return Error::InvalidArgument;
155 }
156
157 const std::uint8_t sw1 = rapdu[rapdu_len - 2U];
158 const std::uint8_t sw2 = rapdu[rapdu_len - 1U];
159
160 if (rapdu_len < (kMacFieldLen + 2U)) {
161 if (out_cap < 2U) {
162 return Error::BufferTooSmall;
163 }
164 out[0] = sw1;
165 out[1] = sw2;
166 *out_len = 2U;
167 crypto::IncCommandCounter(st.counter);
168 return Error::Ok;
169 }
170
171 crypto::Cmac cm{};
172 crypto::CmacStart(cm, st.srmac);
173 crypto::CmacUpdate(cm, st.mcv, 16U);
174 crypto::CmacUpdate(cm, rapdu, rapdu_len - kMacFieldLen - 2U);
175 crypto::CmacUpdate(cm, rapdu + rapdu_len - 2U, 2U);
176 std::uint8_t mac_full[16]{};
177 crypto::CmacFinish(cm, mac_full);
178 if (std::memcmp(mac_full, rapdu + rapdu_len - kMacFieldLen - 2U, kMacFieldLen) != 0) {
179 st.open = false;
180 return Error::Protocol;
181 }
182
183 const std::size_t enc_len = rapdu_len - kMacFieldLen - 2U;
184 if (enc_len == 0U) {
185 if (out_cap < 2U) {
186 return Error::BufferTooSmall;
187 }
188 out[0] = sw1;
189 out[1] = sw2;
190 *out_len = 2U;
191 crypto::IncCommandCounter(st.counter);
192 return Error::Ok;
193 }
194 if ((enc_len % 16U) != 0U || enc_len > out_cap) {
195 st.open = false;
196 return Error::Protocol;
197 }
198
199 std::uint8_t icv[16]{};
200 crypto::ResponseIcv(st.senc, st.counter, icv);
201 if (!crypto::Aes128CbcDecrypt(st.senc, icv, rapdu, out, enc_len)) {
202 st.open = false;
203 return Error::Protocol;
204 }
205 std::size_t plain_len = 0;
206 if (!crypto::IsoUnpad80(out, enc_len, &plain_len)) {
207 st.open = false;
208 return Error::Protocol;
209 }
210 if ((plain_len + 2U) > out_cap) {
211 return Error::BufferTooSmall;
212 }
213 out[plain_len] = sw1;
214 out[plain_len + 1U] = sw2;
215 *out_len = plain_len + 2U;
216 crypto::IncCommandCounter(st.counter);
217 return Error::Ok;
218}

236inline bool FillPutKeyBlock(std::uint8_t* block, const std::uint8_t current_dek[crypto::kAes128KeyLen],
237 const std::uint8_t plain[crypto::kAes128KeyLen],
238 std::uint8_t kcv3[kPutKeyKcvLen]) noexcept {
239 if (block == nullptr || current_dek == nullptr || plain == nullptr || kcv3 == nullptr) {
240 return false;
241 }
242 block[0] = kPutKeyTypeAes;
243 block[1] = static_cast<std::uint8_t>(crypto::kAes128KeyLen + 1U);
244 block[2] = static_cast<std::uint8_t>(crypto::kAes128KeyLen);
245 crypto::Aes128EncryptBlock(current_dek, plain, block + 3U);
246 std::uint8_t ones[crypto::kAes128KeyLen];
247 std::memset(ones, 0x01, sizeof(ones));
248 std::uint8_t kcv_full[crypto::kAes128KeyLen]{};
249 crypto::Aes128EncryptBlock(plain, ones, kcv_full);
250 block[3U + crypto::kAes128KeyLen] = kPutKeyKcvLen;
251 std::memcpy(block + 3U + crypto::kAes128KeyLen + 1U, kcv_full, kPutKeyKcvLen);
252 std::memcpy(kcv3, kcv_full, kPutKeyKcvLen);
253 return true;
254}

264[[nodiscard]] inline Error BuildPutKey(std::uint8_t key_version,
265 const std::uint8_t current_dek[crypto::kAes128KeyLen],
266 const std::uint8_t new_enc[crypto::kAes128KeyLen],
267 const std::uint8_t new_mac[crypto::kAes128KeyLen],

270 std::uint8_t* expect_rsp, std::size_t expect_cap,

273 return Error::InvalidArgument;
274 }
275 *capdu_len = 0;

278 capdu == nullptr || expect_rsp == nullptr) {
279 return Error::InvalidArgument;
280 }

282 return Error::BufferTooSmall;

290 return Error::InvalidArgument;

293 kcv_off += kPutKeyKcvLen;
294 if (!FillPutKeyBlock(data + off, current_dek, new_mac, expect_rsp + kcv_off)) {
295 return Error::InvalidArgument;
296 }
297 off += kPutKeyBlockBytes;
298 kcv_off += kPutKeyKcvLen;
299 if (!FillPutKeyBlock(data + off, current_dek, new_dek, expect_rsp + kcv_off)) {
300 return Error::InvalidArgument;
301 }
302 *expect_len = kPutKeyExpectRspBytes;
303 return apdu::BuildCaseShort(kClaGp, kInsPutKey, key_version, kPutKeyP2Multiple, data,
304 static_cast<std::uint8_t>(kPutKeyDataBytes), false, 0x00U, capdu, capdu_cap,

323 [[nodiscard]] Error OpenSecureChannel(T1Session<TransportT>& t1, const StaticKeys& keys,
324 std::uint32_t timeout_ms) noexcept {
325 using crypto::kAes128KeyLen;
326 using crypto::kChallengeLen;
327 using crypto::kCryptogramLen;
328 st_.open = false;
329 if (!crypto::StaticKeysUsable(keys.host_static_key_enc, keys.host_static_key_enc_len,
330 keys.host_static_key_cmac, keys.host_static_key_cmac_len) ||
331 keys.host_challenge == nullptr || keys.host_challenge_len != kChallengeLen) {
332 return Error::NotSupported;
333 }
334
335 std::uint8_t capdu[16]{};
336 std::size_t capdu_len = 0;
337 Error e = apdu::BuildCaseShort(kClaGp, kInsInitializeUpdate, keys.key_version, 0x00U,
338 keys.host_challenge, static_cast<std::uint8_t>(kChallengeLen),
339 false, 0x00U, capdu, sizeof(capdu), &capdu_len);
340 if (e != Error::Ok) {
341 return e;
342 }
343
344 std::uint8_t rapdu[48]{};
345 std::size_t rapdu_len = 0;
346 e = t1.ExchangeInformation(capdu, capdu_len, rapdu, sizeof(rapdu), &rapdu_len, timeout_ms);
347 if (e != Error::Ok) {
348 return e;
349 }
350 if (rapdu_len < kInitUpdateRspMin) {
351 return Error::Protocol;
352 }
353 const std::uint8_t sw1 = rapdu[rapdu_len - 2U];
354 const std::uint8_t sw2 = rapdu[rapdu_len - 1U];
355 if (sw1 != 0x90U || sw2 != 0x00U) {
356 return Error::Protocol;
357 }
358
359 const std::uint8_t* card_challenge = rapdu + 13U;
360 const std::uint8_t* card_cryptogram = rapdu + 21U;
361
362 std::uint8_t senc[kAes128KeyLen]{};
363 std::uint8_t smac[kAes128KeyLen]{};
364 std::uint8_t srmac[kAes128KeyLen]{};
365 if (!crypto::DeriveSessionKey(keys.host_static_key_enc, crypto::kDeriveSenc,
366 keys.host_challenge, card_challenge, senc) ||
367 !crypto::DeriveSessionKey(keys.host_static_key_cmac, crypto::kDeriveSmac,
368 keys.host_challenge, card_challenge, smac) ||
369 !crypto::DeriveSessionKey(keys.host_static_key_cmac, crypto::kDeriveSrmac,
370 keys.host_challenge, card_challenge, srmac)) {
371 return Error::InvalidArgument;
372 }
373
374 std::uint8_t expect_card[kCryptogramLen]{};
375 if (!crypto::ComputeCryptogram(smac, crypto::kDeriveCardCryptogram, keys.host_challenge,
376 card_challenge, expect_card) ||
377 std::memcmp(expect_card, card_cryptogram, kCryptogramLen) != 0) {
378 return Error::Protocol;
379 }
380
381 std::uint8_t host_cryptogram[kCryptogramLen]{};
382 if (!crypto::ComputeCryptogram(smac, crypto::kDeriveHostCryptogram, keys.host_challenge,
383 card_challenge, host_cryptogram)) {
384 return Error::InvalidArgument;
385 }
386
387 std::uint8_t ext[32]{};
388 ext[0] = kClaGpSecure;
389 ext[1] = kInsExternalAuthenticate;
390 ext[2] = kSecLevelCmacCencRmacRenc;
391 ext[3] = 0x00U;
392 ext[4] = 0x10U;
393 std::memcpy(ext + 5, host_cryptogram, kCryptogramLen);
394
395 std::uint8_t mcv[16]{};
396 std::uint8_t mac_in[16 + 13]{};
397 std::memcpy(mac_in, mcv, 16);
398 std::memcpy(mac_in + 16, ext, 13);
399 std::uint8_t mac_full[16]{};
400 crypto::Aes128Cmac(smac, mac_in, 16U + 13U, mac_full);
401 std::memcpy(ext + 13, mac_full, kCryptogramLen);
402 const std::size_t ext_len = 5U + 2U * kCryptogramLen;
403
404 std::uint8_t ext_rapdu[16]{};
405 std::size_t ext_rapdu_len = 0;
406 e = t1.ExchangeInformation(ext, ext_len, ext_rapdu, sizeof(ext_rapdu), &ext_rapdu_len,
407 timeout_ms);
408 if (e != Error::Ok) {
409 return e;
410 }
411 if (ext_rapdu_len < 2U || ext_rapdu[ext_rapdu_len - 2U] != 0x90U ||
412 ext_rapdu[ext_rapdu_len - 1U] != 0x00U) {
413 return Error::Protocol;
414 }
415
416 std::memcpy(st_.senc, senc, kAes128KeyLen);
417 std::memcpy(st_.smac, smac, kAes128KeyLen);
418 std::memcpy(st_.srmac, srmac, kAes128KeyLen);
419 std::memcpy(st_.mcv, mac_full, 16);
420 std::memset(st_.counter, 0, sizeof(st_.counter));
421 st_.counter[15] = 0x01U;
422 st_.open = true;
423 return Error::Ok;
424 }

428 return Error::Ok;
429 }
