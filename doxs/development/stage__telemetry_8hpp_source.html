73inline int SampleAdcMillivolts(adc_oneshot_unit_handle_t handle)
74{
75 int acc = 0;
76 int raw = 0;
77 for (std::size_t i = 0; i < kOversampleCount; ++i) {
78 if (adc_oneshot_read(handle, static_cast<adc_channel_t>(kAdcChannel), &raw) == ESP_OK) {
79 acc += raw;
80 }
81 }
82 const int avg = acc / static_cast<int>(kOversampleCount);
83 // Rough mV approximation for 12-bit ADC @ 11 dB attenuation on ESP32-S3.
84 // Replace with calibration-based conversion for clinical accuracy.
85 return (avg * 3300) / 4095;
86}

117template <class DeviceT>
118inline bool SignPayload(DeviceT& chip,
119 const char* payload, std::size_t len,
120 std::uint8_t* sig, std::size_t cap, std::size_t* sig_len)
121{
122 std::array<std::uint8_t, 32> digest{};
123 (void)mbedtls_sha256(reinterpret_cast<const std::uint8_t*>(payload), len,
124 digest.data(), /*is224=*/0);
125
126 const se050::Error sg = chip.EcdsaSign(slot::kDeviceIdentityKey,
127 se050::cmd::EcdsaAlgo::Sha256,
128 digest.data(), digest.size(),
129 sig, cap, sig_len, 500U);
130 return sg == se050::Error::Ok;
131}

165template <class DeviceT>
166inline void RunStage(DeviceT& chip, std::uint32_t iterations = 5U)
167{
168 ESP_LOGI(kTag, "==================== STAGE 4 — TELEMETRY LOOP ==================");
169
170#if HF_SE050_LIFECYCLE_ENABLE_NETWORK
171 // -- ADC setup ------------------------------------------------------------
172 adc_oneshot_unit_init_cfg_t unit_cfg{};
173 unit_cfg.unit_id = ADC_UNIT_1;
174 unit_cfg.ulp_mode = ADC_ULP_MODE_DISABLE;
175
176 adc_oneshot_unit_handle_t adc_handle = nullptr;
177 if (adc_oneshot_new_unit(&unit_cfg, &adc_handle) != ESP_OK) {
178 ESP_LOGE(kTag, "ADC unit init failed.");
179 return;
180 }
181 adc_oneshot_chan_cfg_t chan_cfg{};
182 chan_cfg.atten = ADC_ATTEN_DB_12; // 0..~3.1 V input range on S3.
183 chan_cfg.bitwidth = ADC_BITWIDTH_DEFAULT;
184 (void)adc_oneshot_config_channel(adc_handle,
185 static_cast<adc_channel_t>(kAdcChannel), &chan_cfg);
186#endif
187
188 char topic[128]{};
189 (void)std::snprintf(topic, sizeof(topic), "%s/%s/data",
190 aws::kTelemetryTopicPrefix, aws::kDefaultThingName);
191
192 for (std::uint32_t n = 0; n < iterations; ++n) {
193 // -- 1) sample sensor -----------------------------------------------
194#if HF_SE050_LIFECYCLE_ENABLE_NETWORK
195 const int mv = SampleAdcMillivolts(adc_handle);
196#else
197 const int mv = 1234 + static_cast<int>(n); // synthetic
198#endif
199
200 // -- 2) format payload ---------------------------------------------
201 char payload[160]{};
202 const std::size_t plen = FormatPayload(payload, sizeof(payload),
203 /*t_ms=*/static_cast<std::uint64_t>(n) * kSamplePeriodMs,
204 /*counter=*/n,
205 /*mv=*/mv,
206 /*thing=*/aws::kDefaultThingName);
207
208#if HF_SE050_LIFECYCLE_SIGN_EVERY_MESSAGE
209 // -- 3) sign with SE050 --------------------------------------------
210 std::uint8_t sig[96]{};
211 std::size_t sig_len = 0;
212 const bool signed_ok = SignPayload(chip, payload, plen, sig, sizeof(sig), &sig_len);
213 if (!signed_ok) {
214 ESP_LOGE(kTag, "SE050 payload-sign failed; dropping message.");
215 continue;
216 }
217#else
218 std::uint8_t sig[1]{};
219 std::size_t sig_len = 0;
220#endif
221
222 // -- 4) publish (stub today; real mqtt call once linked) ------------
223 PublishStub(topic, payload, plen, sig, sig_len);
224
225#if HF_SE050_LIFECYCLE_ENABLE_NETWORK
226 vTaskDelay(pdMS_TO_TICKS(kSamplePeriodMs));
227#endif
228 }
229
230#if HF_SE050_LIFECYCLE_ENABLE_NETWORK
231 (void)adc_oneshot_del_unit(adc_handle);
232#endif
233
234 ESP_LOGI(kTag, "==================== STAGE 4 — CYCLE DONE ======================");
235}
