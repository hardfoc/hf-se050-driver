78template <class DeviceT>
79inline bool HandleCommand(DeviceT& chip,
80 const std::uint8_t* blob, std::size_t blob_len)
81{
82 if (blob == nullptr || blob_len == 0U) {
83 ESP_LOGW(kTag, "Empty command blob — ignoring.");
84 return false;
85 }
86
87 const auto id = static_cast<CommandId>(blob[0]);
88 const std::uint8_t* payload = blob + 1;
89 const std::size_t plen = blob_len - 1U;
90
91 switch (id) {
92 case CommandId::Ping: {
93 ESP_LOGI(kTag, "Ping received. (Cloud reply should be posted on /resp topic.)");
94 return true;
95 }
96 case CommandId::RequestReprovision: {
97 ESP_LOGW(kTag, "Re-provisioning request received — validating token…");
98 const bool ok = provisioning::RequestReprovisioning(chip, payload, plen);
99 ESP_LOGW(kTag, "Reprovision verdict: %s", ok ? "ACCEPTED (reboot required)" : "REJECTED");
100 return ok;
101 }
102 case CommandId::OtaManifest: {
103 ESP_LOGI(kTag, "OTA manifest received (%u B) — hand off to stage_ota_verify.",
104 static_cast<unsigned>(plen));
105 // TODO(integration): call hf_se050_lifecycle::ota::VerifyManifest(...)
106 // with the embedded image hash + signature, then, if true, kick
107 // esp_https_ota_begin() / your MCU's OTA path.
108 return true;
109 }
110 case CommandId::UpdateConfig: {
111 ESP_LOGI(kTag, "Config update (%u B) — writing to slot::kProvisioningConfig.",
112 static_cast<unsigned>(plen));
113 // TODO(integration): verify outer signature using
114 // kReprovisionAuthorityKey before committing.
115 const se050::Error we = chip.WriteBinary(slot::kProvisioningConfig,
116 payload, plen,
117 /*update=*/true, 0U, true,
118 static_cast<std::uint16_t>(plen), 500U);
119 return we == se050::Error::Ok;
120 }
121 default:
122 ESP_LOGW(kTag, "Unknown command id 0x%02X — dropping.",
123 static_cast<unsigned>(blob[0]));
124 return false;
125 }
126}
